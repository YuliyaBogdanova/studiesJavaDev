-- Создание схемы (если нужно)
CREATE SCHEMA IF NOT EXISTS world_geo;

-- Таблица континентов
CREATE TABLE world_geo.continents (
    continent_id SERIAL PRIMARY KEY,
    continent_name VARCHAR(50) NOT NULL,
    area_sq_km DECIMAL(15,2),
    population BIGINT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Таблица стран
CREATE TABLE world_geo.countries (
    country_id SERIAL PRIMARY KEY,
    country_name VARCHAR(100) NOT NULL,
    country_code CHAR(3) UNIQUE,
    capital VARCHAR(100),
    population BIGINT,
    area_sq_km DECIMAL(15,2),
    continent_id INTEGER,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_continent 
        FOREIGN KEY (continent_id) 
        REFERENCES world_geo.continents(continent_id)
        ON DELETE SET NULL
);

-- Таблица городов
CREATE TABLE world_geo.cities (
    city_id SERIAL PRIMARY KEY,
    city_name VARCHAR(100) NOT NULL,
    country_id INTEGER,
    population INTEGER,
    foundation_year INTEGER,
    is_capital BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_country 
        FOREIGN KEY (country_id) 
        REFERENCES world_geo.countries(country_id)
        ON DELETE CASCADE,
    CONSTRAINT check_foundation_year 
        CHECK (foundation_year <= EXTRACT(YEAR FROM CURRENT_DATE))
);

-- Таблица языков
CREATE TABLE world_geo.languages (
    language_id SERIAL PRIMARY KEY,
    language_name VARCHAR(50) NOT NULL,
    language_family VARCHAR(50),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Таблица связи стран и языков (многие ко многим)
CREATE TABLE world_geo.country_languages (
    country_id INTEGER,
    language_id INTEGER,
    is_official BOOLEAN DEFAULT FALSE,
    percentage DECIMAL(5,2),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_country_languages 
        PRIMARY KEY (country_id, language_id),
    CONSTRAINT fk_country 
        FOREIGN KEY (country_id) 
        REFERENCES world_geo.countries(country_id)
        ON DELETE CASCADE,
    CONSTRAINT fk_language 
        FOREIGN KEY (language_id) 
        REFERENCES world_geo.languages(language_id)
        ON DELETE CASCADE,
    CONSTRAINT check_percentage 
        CHECK (percentage >= 0 AND percentage <= 100)
);

-- Таблица достопримечательностей
CREATE TABLE world_geo.landmarks (
    landmark_id SERIAL PRIMARY KEY,
    landmark_name VARCHAR(100) NOT NULL,
    city_id INTEGER,
    description TEXT,
    year_built INTEGER,
    type VARCHAR(20) CHECK (type IN ('Historical', 'Natural', 'Cultural', 'Religious', 'Modern', 'Monument')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_city 
        FOREIGN KEY (city_id) 
        REFERENCES world_geo.cities(city_id)
        ON DELETE SET NULL,
    CONSTRAINT check_year_built 
        CHECK (year_built <= EXTRACT(YEAR FROM CURRENT_DATE))
);

-- Создание индексов для улучшения производительности
CREATE INDEX idx_countries_continent ON world_geo.countries(continent_id);
CREATE INDEX idx_cities_country ON world_geo.cities(country_id);
CREATE INDEX idx_landmarks_city ON world_geo.landmarks(city_id);
CREATE INDEX idx_country_languages_country ON world_geo.country_languages(country_id);
CREATE INDEX idx_country_languages_language ON world_geo.country_languages(language_id);

-- Функция для обновления поля updated_at
CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Триггеры для автоматического обновления updated_at
CREATE TRIGGER update_continents_modtime
BEFORE UPDATE ON world_geo.continents
FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER update_countries_modtime
BEFORE UPDATE ON world_geo.countries
FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER update_cities_modtime
BEFORE UPDATE ON world_geo.cities
FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER update_languages_modtime
BEFORE UPDATE ON world_geo.languages
FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER update_country_languages_modtime
BEFORE UPDATE ON world_geo.country_languages
FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER update_landmarks_modtime
BEFORE UPDATE ON world_geo.landmarks
FOR EACH ROW EXECUTE FUNCTION update_modified_column();
